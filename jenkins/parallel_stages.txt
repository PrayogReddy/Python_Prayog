pipeline {
    agent { label 'linux' }
    environment {
        sonar_url = "https://sonar.opqtech.com"
        jfrog_url = "https://jfrog.opqtech.com"
    }
    parameters {
        string(name: 'VERSION', defaultValue: '1.0.0', description: 'Version of the application')
        booleanParam(name: 'RUN_SONAR', defaultValue: false, description: 'Run SonarQube scan')
        choice(name: 'environment', choices: ['dev', 'qa', 'prod'], description: 'Deployment environment')
        choice(name: 'region', choices: ['us-east-1', 'us-west-1', 'ap-southeast-1'], description: 'Deployment region')
    }

    options {
        buildDiscarder logRotator(artifactDaysToKeepStr: '', artifactNumToKeepStr: '', daysToKeepStr: '', numToKeepStr: '10')
        disableConcurrentBuilds abortPrevious: true
        timeout(time: 120, unit: 'SECONDS')
        disableRestartFromStage()
    }

    stages {
        stage('Checkout') {
            steps {
                checkout scmGit(branches: [[name: '*/main']], extensions: [], userRemoteConfigs: [[credentialsId: 'jenkins-pvt-key', url: 'https://github.com/OpqTech/java-onlinebookstore.git']])
            }
        }
        stage('Build') {
            steps {
                echo "Build URL: ${env.BUILD_URL}"
                sh 'mvn clean package -DskipTests'
                sh "mv target/*.war target/onlinebookstore-${params.VERSION}.war"
                archiveArtifacts artifacts: 'target/*.war, target/*.exec'
            }
        }
        stage('Test') {
            steps {
                sh 'mvn test'
            }
        }
        stage('Scan') {
            when {
                expression { params.RUN_SONAR == true }
            }
            steps {
                echo "Scan stage"
                echo "SonarQube URL: ${sonar_url}"
            }
        }
        stage('Push') {
            steps {
                echo "Push stage"
                echo "JFrog URL: ${jfrog_url}"
            }
        }
        stage('Deploy') { 
            steps {
                echo "Deploy stage"
                echo "Environment: ${params.environment}"
                echo "Region: ${params.region}"
            }
        }
        stage('Automation suite') {
            parallel {
                stage('suite1') {
                    steps {
                        sh 'sleep 10s'
                        sh 'echo suite1 is success'
                    }
                }
                stage('suite2') {
                    steps {
                        sh 'sleep 20s'
                        sh 'echo suite2 is success'
                    }
                }
                stage('suite3') {
                    steps {
                        sh 'sleep 10s'
                        sh 'echo suite3 is success'
                    }
                }
            }
        }
    }
    post {
        success {
            slackSend(
                message: "BUILD SUCCESS: Job '${env.JOB_NAME}' [Build #${env.BUILD_NUMBER}] completed successfully! (Version: ${params.APP_VERSION}) \n Console Logs: ${env.BUILD_URL}"
            )
        }
        failure {
            slackSend(
                message: "BUILD FAILED: Job '${env.JOB_NAME}' [Build #${env.BUILD_NUMBER}] failed during execution. \n Review Failures: ${env.BUILD_URL}"
            )
        }
        aborted {
            slackSend(
                message: "BUILD ABORTED: Job '${env.JOB_NAME}' [Build #${env.BUILD_NUMBER}] was stopped manually. (${env.BUILD_URL})"
            )
        }
        always {
            cleanWs()
        }
    }
}
